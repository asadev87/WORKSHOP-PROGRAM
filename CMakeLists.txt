cmake_minimum_required(VERSION 3.16)
project(coworking_reservations LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(mysql-concpp CONFIG REQUIRED)
if(NOT TARGET mysql::concpp-jdbc)
  message(FATAL_ERROR "MySQL Connector/C++ was found without its classic/JDBC component.")
endif()
add_executable(coworking src/main.cpp)
target_link_libraries(coworking PRIVATE mysql::concpp-jdbc)
enable_testing()
add_executable(pricing_test tests/pricing.cpp)
add_test(NAME pricing COMMAND pricing_test)
if(WIN32)
  target_compile_definitions(coworking PRIVATE NOMINMAX)
endif()
